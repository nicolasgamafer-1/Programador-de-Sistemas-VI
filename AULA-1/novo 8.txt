#include <stdio.h>
 
int main() {
    
int funcionario = 0, hora = 0;
    double salario = 0, conta = 0;

    scanf("%d", &funcionario);
    scanf("%d", &hora);
    scanf("%lf", &salario);

    conta = hora * salario;

    printf("NUMBER = %d\n", funcionario);
    printf("SALARY = U$ %.2lf\n", conta);

    return 0;
}
